            const totalPrice = cart.reduce((sum, item) => sum + (item.price * item.quantity), 0);

            // Оновлення лічильника
            if (totalItems > 0) {
                badge.innerText = totalItems;
                badge.classList.remove('hidden');
                cartEmptyView.classList.add('hidden');
                cartFooter.classList.remove('hidden');
            } else {
                badge.classList.add('hidden');
                cartEmptyView.classList.remove('hidden');
                cartFooter.classList.add('hidden');
            }

            totalPriceEl.innerText = `${totalPrice.toLocaleString()} ₴`;

            // Список товарів
            cartItemsContainer.innerHTML = cart.map(item => `
                <div class="flex items-center gap-3 bg-slate-50 p-2.5 rounded-2xl border border-slate-100">
                    <img src="${item.image}" alt="${item.title}" class="w-14 h-14 object-cover rounded-xl bg-white border border-slate-100">
                    <div class="flex-1 min-w-0">
                        <h5 class="text-xs font-semibold text-slate-800 truncate">${item.title}</h5>
                        <p class="text-xs font-bold text-slate-900 mt-0.5">${item.price} ₴</p>
                    </div>
                    <div class="flex items-center gap-1.5 bg-white border border-slate-200 rounded-xl p-1">
                        <button onclick="changeQuantity(${item.id}, -1)" class="w-6 h-6 flex items-center justify-center text-slate-600 hover:bg-slate-100 rounded-lg text-xs font-bold">-</button>
                        <span class="text-xs font-semibold px-1 text-slate-800">${item.quantity}</span>
                        <button onclick="changeQuantity(${item.id}, 1)" class="w-6 h-6 flex items-center justify-center text-slate-600 hover:bg-slate-100 rounded-lg text-xs font-bold">+</button>
                    </div>
                </div>
            `).join('');
        }

        // Зміна кількості товару
        function changeQuantity(productId, delta) {
            const item = cart.find(i => i.id === productId);
            if (!item) return;

            item.quantity += delta;
            if (item.quantity <= 0) {
                cart = cart.filter(i => i.id !== productId);
            }
            updateCartUI();
        }

        // Вибір категорії
        function selectCategory(cat) {
            currentCategory = cat;
            document.getElementById('catalog-title').innerText = cat === "Всі" ? "Усі товари" : `Категорія: ${cat}`;
            renderCategories();
            renderProducts();
        }

        // Пошук
        function handleSearch() {
            searchQuery = document.getElementById('search-input').value;
            renderProducts();
        }

        // Скидання фільтрів
        function resetFilters() {
            currentCategory = "Всі";
            searchQuery = "";
            document.getElementById('search-input').value = "";
            renderCategories();
            renderProducts();
        }

        // Модальне вікно кошика
        function toggleCartModal(show) {
            const modal = document.getElementById('cart-modal');
            const backdrop = document.getElementById('cart-backdrop');
            const panel = document.getElementById('cart-panel');

            if (show) {
                modal.classList.remove('invisible');
                setTimeout(() => {
                    backdrop.classList.add('opacity-100');
                    panel.classList.remove('translate-x-full');
                }, 10);
            } else {
                backdrop.classList.remove('opacity-100');
                panel.classList.add('translate-x-full');
                setTimeout(() => {
                    modal.classList.add('invisible');
                }, 300);
            }
        }

        // Відкриття модалки оформлення
        function openCheckoutModal(type) {
            if (cart.length === 0) return;

            if (type === 'telegram') {
                // Формування тексту для Telegram
                let text = "Нове замовлення з сайту NovaMarket:%0A%0A";
                cart.forEach(item => {
                    text += `• ${item.title} x${item.quantity} = ${item.price * item.quantity} ₴%0A`;
                });
                const total = cart.reduce((sum, item) => sum + (item.price * item.quantity), 0);
                text += `%0AЗагальна сума: ${total} ₴`;

                // Замість реального переходу відображаємо вікно для контакту або копіюємо
                window.open(`https://t.me/share/url?url=${encodeURIComponent(window.location.href)}&text=${text}`, '_blank');
                toggleCartModal(false);
                showToast("Замовлення сформовано для Telegram!");
            } else {
                toggleCartModal(false);
                toggleCheckoutModal(true);
            }
        }

        function toggleCheckoutModal(show) {
            const modal = document.getElementById('checkout-modal');
            if (show) {
                modal.classList.remove('invisible');
                setTimeout(() => {
                    modal.classList.add('opacity-100');
                    modal.firstElementChild.classList.remove('scale-95');
                }, 10);
            } else {
                modal.classList.remove('opacity-100');
                modal.firstElementChild.classList.add('scale-95');
                setTimeout(() => {
                    modal.classList.add('invisible');
                }, 200);
            }
        }

        // Обробка форми замовлення
        function handleOrderSubmit(event) {
            event.preventDefault();
            toggleCheckoutModal(false);
            cart = [];
            updateCartUI();
            showToast("Дякуємо! Замовлення успішно прийнято.");
        }

        // Сповіщення (Toast)
        function showToast(message) {
            const toast = document.getElementById('toast');
            const toastMessage = document.getElementById('toast-message');
            toastMessage.innerText = message;

            toast.classList.remove('translate-y-16', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('translate-y-16', 'opacity-0');
            }, 2500);
        }
    </script>
